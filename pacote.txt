const fs = require('node:fs');
const path = require('node:path');
const crypto = require('node:crypto');
const zlib = require('node:zlib');
const folder = process.argv[2];
const text = fs.readFileSync(path.join(folder, 'conteudo.txt'), 'ascii').replace(/\s/g, '');
if (!/^[a-z]+$/.test(text)) throw new Error('Conteudo copiado incorretamente.');
let number = 0n;
for (let i = 0; i < text.length; i += 10) {
  const part = text.slice(i, i + 10);
  let digit = 0;
  for (const c of part) digit = digit * 26 + c.charCodeAt(0) - 97;
  number = number * (26n ** BigInt(part.length)) + BigInt(digit);
}
const hex = number.toString(16).padStart(53668, '0');
const data = Buffer.from(hex, 'hex');
const expected = 'f5a970c3ef7443850e7f71ccf1add42adadccc1740490a01de12058a439be5f2';
if (crypto.createHash('sha256').update(data).digest('hex') !== expected) {
  throw new Error('O texto foi copiado incompleto ou alterado. Copie novamente todo o bloco.');
}
const files = JSON.parse(zlib.brotliDecompressSync(data, { maxOutputLength: 2 * 1024 * 1024 }).toString('utf8'));
for (const [name, content] of Object.entries(files)) {
  if (path.basename(name) !== name || typeof content !== 'string') throw new Error('Arquivo invalido');
  fs.writeFileSync(path.join(folder, name), content, { encoding: 'utf8', flag: 'wx' });
}
fs.writeFileSync(path.join(folder, 'ABRIR-PAINEL.cmd'), '@echo off\r\ncd /d "%~dp0"\r\ncall npm.cmd start\r\npause\r\n', { flag: 'wx' });
console.log('Todos os arquivos da aplicacao foram criados.');
